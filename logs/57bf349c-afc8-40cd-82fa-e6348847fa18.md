import sys


def main():
    if len(sys.argv) != 3:
        sys.exit(2)

    path = sys.argv[1]
    generations = int(sys.argv[2])

    with open(path, "r") as f:
        grid = [list(line) for line in f.read().splitlines()]

    height = len(grid)
    width = len(grid[0]) if height else 0

    for _ in range(generations):
        new_grid = [["." for _ in range(width)] for _ in range(height)]

        for r in range(height):
            for c in range(width):
                live_neighbors = 0

                for dr in (-1, 0, 1):
                    for dc in (-1, 0, 1):
                        if dr == 0 and dc == 0:
                            continue

                        nr = r + dr
                        nc = c + dc

                        if 0 <= nr < height and 0 <= nc < width:
                            if grid[nr][nc] == "#":
                                live_neighbors += 1

                if grid[r][c] == "#":
                    if live_neighbors in (2, 3):
                        new_grid[r][c] = "#"
                else:
                    if live_neighbors == 3:
                        new_grid[r][c] = "#"

        grid = new_grid

    if grid:
        sys.stdout.write("\n".join("".join(row) for row in grid) + "\n")


if __name__ == "__main__":
    main()
